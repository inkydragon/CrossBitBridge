cmake_minimum_required(VERSION 3.24)
project(CrossBitBridgeCppNcalRpc VERSION 0.1.0 LANGUAGES C CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "Build on Windows with MSVC and the Windows SDK (MIDL).")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(NCAL_IDL "${CMAKE_CURRENT_SOURCE_DIR}/spec/sample.idl" CACHE FILEPATH "RPC specification")
set(NCAL_INTERFACE "NcalRpc_v1_0" CACHE STRING "MIDL interface symbol prefix")
set(NCAL_ADAPTER "${CMAKE_CURRENT_SOURCE_DIR}/examples/sample/adapter.cpp" CACHE FILEPATH "Server manager implementation")
set(NCAL_CLIENT "${CMAKE_CURRENT_SOURCE_DIR}/examples/sample/client.cpp" CACHE FILEPATH "Client C ABI implementation")
set(NCAL_PROBE "${CMAKE_CURRENT_SOURCE_DIR}/examples/sample/probe.cpp" CACHE FILEPATH "Client smoke test")
get_filename_component(NCAL_IDL "${NCAL_IDL}" ABSOLUTE BASE_DIR "${CMAKE_SOURCE_DIR}")
if(NOT EXISTS "${NCAL_IDL}")
  message(FATAL_ERROR "IDL does not exist: ${NCAL_IDL}")
endif()
# CMake's Visual Studio generator supplies the SDK tool environment to CustomBuild.
if(CMAKE_SIZEOF_VOID_P EQUAL 4)
  set(midl_env win32)
else()
  set(midl_env x64)
endif()
set(gen "${CMAKE_CURRENT_BINARY_DIR}/generated")
file(MAKE_DIRECTORY "${gen}")
add_custom_command(
  OUTPUT "${gen}/bridge.h" "${gen}/bridge_c.c" "${gen}/bridge_s.c"
  COMMAND midl /nologo /env ${midl_env} /target NT60 /robust /Oicf
    /prefix client Wire_ server Impl_
    /h bridge.h /cstub bridge_c.c /sstub bridge_s.c /out "${gen}" "${NCAL_IDL}"
  DEPENDS "${NCAL_IDL}"
  VERBATIM COMMENT "Generate ${midl_env} RPC stubs from ${NCAL_IDL}")
add_custom_target(rpc_codegen DEPENDS "${gen}/bridge.h" "${gen}/bridge_c.c" "${gen}/bridge_s.c")
function(rpc_target target)
  add_dependencies(${target} rpc_codegen)
  target_include_directories(${target} PRIVATE "${gen}" "${CMAKE_CURRENT_SOURCE_DIR}/rpc")
  target_compile_definitions(${target} PRIVATE UNICODE _UNICODE _WIN32_WINNT=0x0602)
  target_link_libraries(${target} PRIVATE Rpcrt4)
endfunction()
if(CMAKE_SIZEOF_VOID_P EQUAL 4)
  add_executable(ncal_host rpc/server.cpp rpc/memory.cpp "${NCAL_ADAPTER}" "${gen}/bridge_s.c")
  rpc_target(ncal_host)
  target_compile_definitions(ncal_host PRIVATE NCAL_SERVER_INTERFACE=Impl_${NCAL_INTERFACE}_s_ifspec)
  target_link_libraries(ncal_host PRIVATE Advapi32)
  target_include_directories(ncal_host PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../legacy/cpp_legacy")
else()
  add_library(ncal_client SHARED rpc/binding.cpp rpc/memory.cpp "${NCAL_CLIENT}" "${gen}/bridge_c.c")
  rpc_target(ncal_client)
  target_compile_definitions(ncal_client PRIVATE NCAL_CLIENT_BUILD)
  add_executable(ncal_probe "${NCAL_PROBE}")
  target_include_directories(ncal_probe PRIVATE rpc "${CMAKE_CURRENT_SOURCE_DIR}/examples/sample")
  target_link_libraries(ncal_probe PRIVATE ncal_client Rpcrt4)
endif()
